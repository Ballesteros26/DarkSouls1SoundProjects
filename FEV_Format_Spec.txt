typedef enum <uint32> {
    BANK_TYPE_STREAM_FROM_DISK       = 0x00000080,
    BANK_TYPE_LOAD_INTO_MEMORY       = 0x00000002,
    BANK_TYPE_DECOMPRESS_INTO_MEMORY = 0x00000001
} BANK_TYPE;

typedef struct {
    BANK_TYPE bank_type;
    uint32    bank_max_streams;
    uint64    hash;             /* "Truncated MD5 Hash -- generated using only information which would break FSB/FEV compat"
                                    This appears to be the first 8 bytes of the MD5 hash of the string generated by the list of sounds
                                    in the wavebank. Each sound is listed in its wavebank order, with the following information:
                                     char   filename[]    (NUL-terminated)
                                     byte   simple_event  (0x00 = No, 0x02 = Yes)
                                     byte   complex_event (0x00 = No, 0x01 = Yes)
                                     uint32 padding       (seems to be always 0x00000000)
                                    Because this data is determined elsewhere in the .fev, it should match automatically if the rest
                                    of the file does. */
    uint32    bank_name_length;
    char      bank_name[bank_name_length];
} WAVEBANK;

typedef struct {
    int32   parent_index;               /* Indexes the array of envelopes to determine the effect information. -1 if this envelope has the effect information. */
    uint32  name_length;
    char    name[name_length];          /* If effect_type == ENVELOPE_EFFECT_TYPE_DSP_EFFECT, then this holds the name of the DSP Effect. Otherwise, it is ''. */
    uint32  effect_parameter_index;     /* Determines which parameter of the effect this envelope controls. */
    uint16  _effect_type_and_muted;
    
    uint16                  muted       = (_effect_type_and_muted & 0x0001)
    ENVELOPE_EFFECT_TYPE    effect_type = (ENVELOPE_EFFECT_TYPE) (_effect_type_and_muted & ~0x0001)
    
    uint16  flags;
    uint32  padding;                    /* Seems to be always 0x00000000. */
    uint32  num_of_points;
    POINT   points[num_of_points];
    uint32  control_parameter_index;    /* Indexes layer's array of parameters to determine which controls this envelope. */
    uin32   mapping_method;             /* Appears to be determined by the effect type. */
} ENVELOPE;

    int32                               steal_priority;
    EVENT_FLAGS                         flags;
    float                               min_distance_3d;
    float                               max_distance_3d;

    EVENT_ONESHOT_TYPE                  oneshot;
    EVENT_PITCH_RAND_UNITS_TYPE         pitch_rand_units;                                               /* Seems unnecessary; pitch_randomization is actually always in quad-octaves. */
    float                               speaker_l_2d_field_ratio;                                       /* See Field Ratio -> dB conversion note at top. */
    float                               speaker_r_2d_field_ratio;                                       /* See Field Ratio -> dB conversion note at top. */
    float                               speaker_c_2d_field_ratio;                                       /* See Field Ratio -> dB conversion note at top. */
    float                               speaker_lfe_field_ratio;                                        /* See Field Ratio -> dB conversion note at top. */
    float                               speaker_lr_2d_field_ratio;                                      /* See Field Ratio -> dB conversion note at top. */
    float                               speaker_rr_2d_field_ratio;                                      /* See Field Ratio -> dB conversion note at top. */
    float                               speaker_ls_2d_field_ratio;                                      /* See Field Ratio -> dB conversion note at top. */
    float                               speaker_rs_2d_field_ratio;                                      /* See Field Ratio -> dB conversion note at top. */
    float                               cone_inside_angle_3d;                                           /* In degrees. */
    float                               cone_outside_angle_3d;                                          /* In degrees. */
    float                               cone_outside_volume_3d_field_ratio;                             /* See Field Ratio -> dB conversion note at top. */
    EVENT_MAX_PLAYBACK_BEHAVIOR_TYPE    max_playback_behavior;
    float                               doppler_factor_3d;
    float                               reverb_dry_level;
    float                               reverb_wet_level;

    float                               spawn_intensity_randomization;
    float                               pan_level_3d;
    uint32                              position_3d_randomization;
    uint32                              num_of_layers;
    if (type == EVENT_TYPE_SIMPLE) {
        SIMPLE_LAYER                    layers[num_of_layers];
    } else {
        COMPLEX_LAYERS                  layers[num_of_layers];
    }
    if (type == EVENT_TYPE_COMPLEX) {
        uint32                          num_of_parameters;
        PARAMETER                       parameters[num_of_parameters];
        uint32                          num_of_user_properties;

}
